// ============================================================ // StockPro — Google Apps Script Backend // วิธีติดตั้ง: // 1. เปิด Google Sheets ใหม่ (หรือที่มีอยู่) // 2. ไปที่ Extensions > Apps Script // 3. วางโค้ดทั้งหมดนี้แทนที่โค้ดเดิม // 4. กด Save แล้วกด Deploy > New Deployment // 5. เลือก Type: Web App // Execute as: Me // Who has access: Anyone // 6. กด Deploy แล้วคัดลอก URL ที่ได้ // 7. นำ URL ไปใส่ในโปรแกรม StockPro // ============================================================ const SHEET_NAME_PRODUCTS = 'Products'; const SHEET_NAME_SALES = 'Sales'; const SHEET_NAME_USERS = 'Users'; // ---------- Bootstrap: สร้าง Sheet ถ้ายังไม่มี ---------- function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, headers.length) .setBackground('#1e2335') .setFontColor('#ffffff') .setFontWeight('bold'); } return sheet; } function initSheets() { getOrCreateSheet(SHEET_NAME_PRODUCTS, ['id','name','category','qty','reorder','cost','price','barcode','image','updatedAt']); getOrCreateSheet(SHEET_NAME_SALES, ['id','time','date','items','qty','total','seller','createdAt']); getOrCreateSheet(SHEET_NAME_USERS, ['id','username','password','name','role','email','phone','color','avatar','updatedAt']); } // ---------- CORS Helper ---------- function corsResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ---------- Main Entry Point ---------- function doGet(e) { const data = handleRequest(e); const json = data.getContent(); const callback = (e.parameter || {}).callback; if(callback) { return ContentService .createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return data; } function doPost(e) { return handleRequest(e); } function handleRequest(e) { try { initSheets(); const params = e.parameter || {}; // รับข้อมูลจาก POST body หรือ GET ?payload= let postData = {}; if(e.postData && e.postData.contents) { postData = JSON.parse(e.postData.contents); } else if(params.payload) { postData = JSON.parse(decodeURIComponent(params.payload)); } const action = params.action || postData.action; switch(action) { // ---- Products ---- case 'getProducts': return corsResponse({ ok:true, data: readSheet(SHEET_NAME_PRODUCTS) }); case 'saveProducts': return corsResponse({ ok:true, data: writeAll(SHEET_NAME_PRODUCTS, postData.data, productHeaders()) }); case 'upsertProduct': return corsResponse({ ok:true, data: upsertRow(SHEET_NAME_PRODUCTS, postData.data, 'id', productHeaders()) }); case 'deleteProduct': return corsResponse({ ok:true, deleted: deleteRow(SHEET_NAME_PRODUCTS, 'id', postData.id) }); // ---- Sales ---- case 'getSales': return corsResponse({ ok:true, data: readSheet(SHEET_NAME_SALES) }); case 'addSale': return corsResponse({ ok:true, data: appendRow(SHEET_NAME_SALES, postData.data, saleHeaders()) }); // ---- Users ---- case 'getUsers': return corsResponse({ ok:true, data: readSheet(SHEET_NAME_USERS) }); case 'saveUsers': return corsResponse({ ok:true, data: writeAll(SHEET_NAME_USERS, postData.data, userHeaders()) }); case 'upsertUser': return corsResponse({ ok:true, data: upsertRow(SHEET_NAME_USERS, postData.data, 'id', userHeaders()) }); case 'deleteUser': return corsResponse({ ok:true, deleted: deleteRow(SHEET_NAME_USERS, 'id', postData.id) }); // ---- Sync All ---- case 'syncAll': return corsResponse({ ok: true, products: readSheet(SHEET_NAME_PRODUCTS), sales: readSheet(SHEET_NAME_SALES), users: readSheet(SHEET_NAME_USERS), }); // ---- Push All ---- case 'pushAll': writeAll(SHEET_NAME_PRODUCTS, postData.products || [], productHeaders()); writeAll(SHEET_NAME_USERS, postData.users || [], userHeaders()); if (postData.sales && postData.sales.length) { writeAll(SHEET_NAME_SALES, postData.sales, saleHeaders()); } return corsResponse({ ok: true, message: 'Push สำเร็จ' }); default: return corsResponse({ ok: false, error: 'Unknown action: ' + action }); } } catch(err) { return corsResponse({ ok: false, error: err.message }); } } // ---------- Sheet Helpers ---------- function productHeaders() { return ['id','name','category','qty','reorder','cost','price','barcode','image','updatedAt']; } function saleHeaders() { return ['id','time','date','items','qty','total','seller','createdAt']; } function userHeaders() { return ['id','username','password','name','role','email','phone','color','avatar','updatedAt']; } function readSheet(sheetName) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0]; return data.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i] === '' ? null : row[i]; }); return obj; }); } function writeAll(sheetName, rows, headers) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return []; // Clear data rows (keep header) const lastRow = sheet.getLastRow(); if (lastRow > 1) sheet.deleteRows(2, lastRow - 1); if (!rows || rows.length === 0) return []; const values = rows.map(r => headers.map(h => { const v = r[h]; return (v === null || v === undefined) ? '' : v; })); sheet.getRange(2, 1, values.length, headers.length).setValues(values); return rows; } function upsertRow(sheetName, rowData, keyField, headers) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return null; rowData.updatedAt = new Date().toISOString(); const data = sheet.getDataRange().getValues(); const headerRow = data[0]; const keyCol = headerRow.indexOf(keyField); if (keyCol === -1) return null; // Find existing row for (let i = 1; i < data.length; i++) { if (data[i][keyCol] == rowData[keyField]) { const values = headers.map(h => (rowData[h] === null || rowData[h] === undefined) ? '' : rowData[h]); sheet.getRange(i + 1, 1, 1, values.length).setValues([values]); return rowData; } } // Append new const values = headers.map(h => (rowData[h] === null || rowData[h] === undefined) ? '' : rowData[h]); sheet.appendRow(values); return rowData; } function appendRow(sheetName, rowData, headers) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return null; rowData.createdAt = new Date().toISOString(); const values = headers.map(h => (rowData[h] === null || rowData[h] === undefined) ? '' : rowData[h]); sheet.appendRow(values); return rowData; } function deleteRow(sheetName, keyField, keyValue) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return false; const data = sheet.getDataRange().getValues(); const keyCol = data[0].indexOf(keyField); if (keyCol === -1) return false; for (let i = data.length - 1; i >= 1; i--) { if (data[i][keyCol] == keyValue) { sheet.deleteRow(i + 1); return true; } } return false; }
